cmake_minimum_required(VERSION 3.16)
project(cml LANGUAGES C)

include(CTest)
include(GNUInstallDirs)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(CML_BLAS none CACHE STRING "Matrix backend: none, accelerate, or openblas")
set_property(CACHE CML_BLAS PROPERTY STRINGS none accelerate openblas)
option(CML_SANITIZERS "Build with address and undefined-behavior sanitizers" OFF)
option(CML_BUILD_EXAMPLES "Build example programs" ON)
option(CML_BUILD_BENCHMARKS "Build benchmark programs" ON)

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(CML_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

file(GLOB cml_sources CONFIGURE_DEPENDS src/*.c)
list(FILTER cml_sources EXCLUDE REGEX "/main\\.c$")
add_library(cml STATIC ${cml_sources})
add_library(CML::cml ALIAS cml)
target_include_directories(cml PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(cml PUBLIC m)

if(CML_BLAS STREQUAL "accelerate")
  find_library(accelerate_library Accelerate)
  if(NOT accelerate_library)
    message(FATAL_ERROR "Apple Accelerate was not found")
  endif()
  target_compile_definitions(cml PUBLIC CML_CBLAS_ACCELERATE)
  target_link_libraries(cml PUBLIC ${accelerate_library})
elseif(CML_BLAS STREQUAL "openblas")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(openblas REQUIRED IMPORTED_TARGET openblas)
  target_compile_definitions(cml PUBLIC CML_CBLAS_OPENBLAS)
  target_link_libraries(cml PUBLIC PkgConfig::openblas)
elseif(NOT CML_BLAS STREQUAL "none")
  message(FATAL_ERROR "CML_BLAS must be none, accelerate, or openblas")
endif()

add_executable(cml_cli src/main.c)
target_link_libraries(cml_cli PRIVATE CML::cml)
set_target_properties(cml_cli PROPERTIES OUTPUT_NAME cml)

if(BUILD_TESTING)
  file(GLOB tests CONFIGURE_DEPENDS tests/test_*.c)
  list(FILTER tests EXCLUDE REGEX "/test_main\\.c$")
  foreach(source IN LISTS tests)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    target_compile_options(${name} PRIVATE -UNDEBUG)
    target_link_libraries(${name} PRIVATE CML::cml)
    add_test(NAME ${name} COMMAND ${name})
    if(CML_SANITIZERS)
      set_tests_properties(${name} PROPERTIES ENVIRONMENT "ASAN_OPTIONS=allocator_may_return_null=1")
    endif()
  endforeach()
endif()

if(CML_BUILD_EXAMPLES)
  file(GLOB examples CONFIGURE_DEPENDS examples/*.c)
  foreach(source IN LISTS examples)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "${source}")
    target_link_libraries(${name} PRIVATE CML::cml)
  endforeach()
endif()

if(CML_BUILD_BENCHMARKS)
  foreach(name IN ITEMS benchmark_matrix benchmark_training)
    add_executable(${name} "benchmarks/${name}.c")
    target_link_libraries(${name} PRIVATE CML::cml)
  endforeach()
endif()

install(TARGETS cml ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
